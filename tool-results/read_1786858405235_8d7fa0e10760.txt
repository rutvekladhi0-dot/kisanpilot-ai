    58→const defaultAlerts: AlertItem[] = [
    59→  { id: 1, icon: '🌧', text: 'rainExpected', read: false },
    60→  { id: 2, icon: '💧', text: 'monitorSoil', read: false },
    61→  { id: 3, icon: '🐛', text: 'checkPestSigns', read: false },

    65→// ============================================================
    66→// Main App Component
    67→// ============================================================
    68→export default function KisanPilotApp() {
    69→  const initialized = useRef(false);
    70→  const [lang, setLangState] = useState<Language>(() => {
    71→    if (typeof window !== 'undefined') {
    72→      return (localStorage.getItem('kp_lang') as Language) || 'en';
    73→    }
    74→    return 'en';
    75→  });
    76→  const [profile, setProfileState] = useState<FarmerProfile | null>(() => {
    77→    if (typeof window !== 'undefined') {
    78→      try {
    79→        const saved = localStorage.getItem('kp_profile');
    80→        return saved ? JSON.parse(saved) : null;
    81→      } catch { return null; }
    82→    }
    83→    return null;
    84→  });
    85→  const [screen, setScreen] = useState<Screen>(() => {
    86→    if (typeof window !== 'undefined') {
    87→      const savedLang = localStorage.getItem('kp_lang');
    88→      const savedProfile = localStorage.getItem('kp_profile');
    89→      if (savedLang && savedProfile) return 'dashboard';
    90→      if (savedLang) return 'profile';
    91→    }
    92→    return 'language';
    93→  });
    94→  const [alerts, setAlertsState] = useState<AlertItem[]>(() => {
    95→    if (typeof window !== 'undefined') {
    96→      try {
    97→        const saved = localStorage.getItem('kp_alerts');
    98→        return saved ? JSON.parse(saved) : defaultAlerts;
    99→      } catch { return defaultAlerts; }
   100→    }
   101→    return defaultAlerts;
   102→  });
   103→
   104→  const [mounted, setMounted] = useState(false);
   105→
   106→  useEffect(() => {
   107→    if (!initialized.current) {
   108→      initialized.current = true;
   109→      requestAnimationFrame(() => setMounted(true));
   110→    }
   111→  }, []);
   112→
   113→  const t = getTranslations(lang);
   114→
   115→  const setLang = useCallback((l: Language) => {
   116→    setLangState(l);
   117→    localStorage.setItem('kp_lang', l);
   118→  }, []);
   119→
   120→  const setProfile = useCallback((p: FarmerProfile) => {
   121→    setProfileState(p);
   122→    localStorage.setItem('kp_profile', JSON.stringify(p));
   123→  }, []);
   124→
   125→  const navigate = useCallback((s: Screen) => {
   126→    setScreen(s);
   127→    window.scrollTo(0, 0);
   128→  }, []);
   129→
   130→  const setAlerts = useCallback((a: AlertItem[]) => {
   131→    setAlertsState(a);
   132→    localStorage.setItem('kp_alerts', JSON.stringify(a));
   133→  }, []);
   134→
   135→  if (!mounted) {
   136→    return null;
   137→  }
   138→
   139→  return (
   140→    <AppContext.Provider value={{ lang, t, setLang, profile, setProfile, screen, navigate, alerts, setAlerts }}>
   141→      <div className="min-h-screen bg-gradient-to-br from-green-50 via-white to-emerald-50">
   142→        <AnimatePresence mode="wait">
   143→          {screen === 'language' && <LanguageScreen key="language" />}
   144→          {screen === 'profile' && <ProfileScreen key="profile" />}
   145→          {screen === 'dashboard' && <DashboardScreen key="dashboard" />}
   146→          {screen === 'chatbot' && <ChatbotScreen key="chatbot" />}
   147→          {screen === 'myfarm' && <MyFarmScreen key="myfarm" />}
   148→          {screen === 'cropdoctor' && <CropDoctorScreen key="cropdoctor" />}
   149→          {screen === 'weather' && <WeatherScreen key="weather" />}
   150→          {screen === 'insights' && <InsightsScreen key="insights" />}
   151→          {screen === 'alerts' && <AlertsScreen key="alerts" />}
   152→          {screen === 'settings' && <SettingsScreen key="settings" />}
   153→          {screen === 'voicekhata' && <VoiceKhataScreen key="voicekhata" />}
   154→          {screen === 'farmeconomics' && <FarmEconomicsScreen key="farmeconomics" />}
   155→          {screen === 'bajarbhav' && <BajarBhavScreen key="bajarbhav" />}
   156→          {screen === 'riskradar' && <RiskRadarScreen key="riskradar" />}
   157→          {screen === 'simulator' && <SimulatorScreen key="simulator" />}
   158→          {screen === 'pmfby' && <PMFBYScreen key="pmfby" />}
   159→          {screen === 'farmgoals' && <FarmGoalsScreen key="farmgoals" />}
   160→          {screen === 'nextbestaction' && <NextBestActionScreen key="nextbestaction" />}
   161→          {screen === 'phototracker' && <MonthlyPhotoTrackerScreen key="phototracker" />}
   162→          {screen === 'farmmemory' && <FarmMemoryScreen key="farmmemory" />}
   163→          {screen === 'seasonscorecard' && <SeasonScoreCardScreen key="seasonscorecard" />}
   164→        </AnimatePresence>
   165→      </div>
   166→    </AppContext.Provider>
   167→  );
   168→}

   275→// PROFILE SETUP SCREEN
   276→// ============================================================
   277→function ProfileScreen() {
   278→  const { t, navigate, setProfile } = useApp();
   279→  const [name, setName] = useState('');
   280→  const [village, setVillage] = useState('');
   281→  const [farmSize, setFarmSize] = useState('');
   282→  const [mainCrop, setMainCrop] = useState('');
   283→  const [photo, setPhoto] = useState<string | null>(null);
   284→  const [errors, setErrors] = useState<{ name?: string; village?: string }>({});
   285→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   286→
   287→  // Family Members state
   288→  const [familyMembers, setFamilyMembers] = useState<Array<{ id: number; name: string; relation: string; phone: string }>>([]);
   289→  const [showAddMember, setShowAddMember] = useState(false);
   290→  const [memberName, setMemberName] = useState('');
   291→  const [memberRelation, setMemberRelation] = useState('');
   292→  const [memberPhone, setMemberPhone] = useState('');
   293→  const [memberFormErrors, setMemberFormErrors] = useState<{ name?: string; phone?: string }>({});
   294→  const [memberToast, setMemberToast] = useState<string | null>(null);
   295→
   296→  React.useEffect(() => {
   297→    if (typeof window !== 'undefined') {
   298→      const saved = localStorage.getItem('kp_family_members');
   299→      if (saved) {
   300→        try { setFamilyMembers(JSON.parse(saved)); } catch { /* ignore */ }
   301→      }
   302→    }
   303→  }, []);
   304→
   305→  const saveFamilyMembers = (members: Array<{ id: number; name: string; relation: string; phone: string }>) => {
   306→    setFamilyMembers(members);
   307→    localStorage.setItem('kp_family_members', JSON.stringify(members));
   308→  };
   309→
   310→  const handleAddMember = () => {
   311→    const newErrors: { name?: string; phone?: string } = {};
   312→    if (!memberName.trim()) newErrors.name = t.memberName + ' ' + t.nameRequired;
   313→    if (!memberPhone.trim()) newErrors.phone = t.memberPhone + ' ' + t.nameRequired;
   314→    if (Object.keys(newErrors).length > 0) {
   315→      setMemberFormErrors(newErrors);
   316→      return;
   317→    }
   318→    const newMember = {
   319→      id: Date.now(),
   320→      name: memberName.trim(),
   321→      relation: memberRelation || t.relationOther,
   322→      phone: memberPhone.trim(),
   323→    };
   324→    const updated = [...familyMembers, newMember];
   325→    saveFamilyMembers(updated);
   326→    setMemberName('');
   327→    setMemberRelation('');
   328→    setMemberPhone('');
   329→    setMemberFormErrors({});
   330→    setShowAddMember(false);
   331→    setMemberToast(t.memberAdded);
   332→    setTimeout(() => setMemberToast(null), 2000);
   333→  };
   334→
   335→  const handleRemoveMember = (id: number) => {
   336→    const updated = familyMembers.filter(m => m.id !== id);
   337→    saveFamilyMembers(updated);
   338→    setMemberToast(t.memberRemoved);
   339→    setTimeout(() => setMemberToast(null), 2000);
   340→  };
   341→
   342→  const maskPhone = (phone: string) => {
   343→    if (phone.length <= 4) return '****' + phone.slice(-4);
   344→    return '****' + phone.slice(-4);
   345→  };
   346→
   347→  const relations = [
   348→    { value: 'Wife', label: t.relationWife },
   349→    { value: 'Son', label: t.relationSon },
   350→    { value: 'Daughter', label: t.relationDaughter },
   351→    { value: 'Father', label: t.relationFather },
   352→    { value: 'Mother', label: t.relationMother },
   353→    { value: 'Brother', label: t.relationBrother },
   354→    { value: 'Other', label: t.relationOther },
   355→  ];
   356→
   357→  const farmSizes = [
   358→    { value: '<1', label: t.lessThanOneAcre },
   359→    { value: '1-5', label: t.oneToFiveAcres },
   360→    { value: '5-10', label: t.fiveToTenAcres },
   361→    { value: '>10', label: t.moreThanTenAcres },
   362→  ];
   363→
   364→  const crops = [t.cropWheat, t.cropRice, t.cropCotton, t.cropSoybean, t.cropSugarcane, t.cropTomato, t.cropOnion, t.cropOther];
   365→
   366→  const handleSave = () => {
   367→    const newErrors: { name?: string; village?: string } = {};
   368→    if (!name.trim()) newErrors.name = t.nameRequired;
   369→    if (!village.trim()) newErrors.village = t.villageRequired;
   370→    if (Object.keys(newErrors).length > 0) {
   371→      setErrors(newErrors);
   372→      return;
   373→    }
   374→    setProfile({
   375→      name: name.trim(),
   376→      village: village.trim(),
   377→      farmSize: farmSize || t.lessThanOneAcre,
   378→      mainCrop: mainCrop || t.cropWheat,
   379→    });
   380→    if (photo) {
   381→      localStorage.setItem('kp_farmer_photo', photo);
   382→    }
   383→    navigate('dashboard');
   384→  };
   385→
   386→  const handlePhotoSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   387→    const file = e.target.files?.[0];
   388→    if (file) {
   389→      const reader = new FileReader();
   390→      reader.onload = (ev) => {
   391→        setPhoto(ev.target?.result as string);
   392→      };
   393→      reader.readAsDataURL(file);
   394→    }
   395→  };
   396→
   397→  const removePhoto = () => {
   398→    setPhoto(null);
   399→    localStorage.removeItem('kp_farmer_photo');
   400→    if (fileInputRef.current) fileInputRef.current.value = '';
   401→  };
   402→
   403→  return (
   404→    <PageWrapper className="flex items-center justify-center px-4 py-8">
   405→      <div className="w-full max-w-md">
   406→        <div className="text-center mb-8">
   407→          <motion.div
   408→            whileHover={{ scale: 1.05 }}
   409→            onClick={() => fileInputRef.current?.click()}
   410→            className="w-24 h-24 rounded-full bg-gradient-to-br from-green-600 to-emerald-700 flex items-center justify-center shadow-lg mx-auto mb-3 cursor-pointer overflow-hidden border-4 border-white shadow-green-200"
   411→          >
   412→            {photo ? (
   413→              <img src={photo} alt="Profile" className="w-full h-full object-cover" />
   414→            ) : (
   415→              <span className="text-4xl">👨‍🌾</span>
   416→            )}
   417→          </motion.div>
   418→          <div className="flex items-center justify-center gap-3 mb-1">
   419→            <button
   420→              onClick={() => fileInputRef.current?.click()}
   421→              className="text-xs text-green-600 font-semibold hover:text-green-800"
   422→            >
   423→              {t.changePhoto}
   424→            </button>
   425→            {photo && (
   426→              <button
   427→                onClick={removePhoto}
   428→                className="text-xs text-red-500 font-semibold hover:text-red-700"
   429→              >
   430→                {t.removePhoto}
   431→              </button>
   432→            )}
   433→          </div>
   434→          <p className="text-xs text-gray-400">{t.uploadPhotoLabel}</p>
   435→          <input
   436→            ref={fileInputRef}
   437→            type="file"
   438→            accept="image/*"
   439→            capture="user"
   440→            onChange={handlePhotoSelect}
   441→            className="hidden"
   442→          />
   443→          <h1 className="text-2xl font-bold text-gray-800 mt-4">{t.setupProfile}</h1>
   444→          <p className="text-gray-500 mt-1">{t.setupProfileSubtitle}</p>
   445→        </div>
   446→
   447→        <div className="bg-white rounded-3xl shadow-lg p-6 md:p-8 space-y-5">
   448→          {/* Farmer Name */}
   449→          <div>
   450→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.farmerName}</label>
   451→            <input
   452→              type="text"
   453→              value={name}
   454→              onChange={(e) => { setName(e.target.value); setErrors(p => ({ ...p, name: undefined })); }}
   455→              placeholder={t.farmerNamePlaceholder}
   456→              className={`w-full px-4 py-3 rounded-xl border-2 ${errors.name ? 'border-red-400 bg-red-50' : 'border-gray-200 bg-gray-50'} focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800`}
   457→            />
   458→            {errors.name && <p className="text-red-500 text-sm mt-1">{errors.name}</p>}
   459→          </div>
   460→
   461→          {/* Village Name */}
   462→          <div>
   463→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.villageName}</label>
   464→            <input
   465→              type="text"
   466→              value={village}
   467→              onChange={(e) => { setVillage(e.target.value); setErrors(p => ({ ...p, village: undefined })); }}
   468→              placeholder={t.villageNamePlaceholder}
   469→              className={`w-full px-4 py-3 rounded-xl border-2 ${errors.village ? 'border-red-400 bg-red-50' : 'border-gray-200 bg-gray-50'} focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800`}
   470→            />
   471→            {errors.village && <p className="text-red-500 text-sm mt-1">{errors.village}</p>}
   472→          </div>
   473→
   474→          {/* Farm Size */}
   475→          <div>
   476→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.farmSize}</label>
   477→            <div className="grid grid-cols-2 gap-2">
   478→              {farmSizes.map((fs) => (
   479→                <button
   480→                  key={fs.value}
   481→                  onClick={() => setFarmSize(fs.label)}
   482→                  className={`px-3 py-2.5 rounded-xl text-sm font-medium border-2 transition-all ${
   483→                    farmSize === fs.label
   484→                      ? 'border-green-500 bg-green-50 text-green-700'
   485→                      : 'border-gray-200 bg-gray-50 text-gray-600 hover:border-gray-300'
   486→                  }`}
   487→                >
   488→                  {fs.label}
   489→                </button>
   490→              ))}
   491→            </div>
   492→          </div>
   493→
   494→          {/* Main Crop */}
   495→          <div>
   496→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.mainCrop}</label>

   904→          if (!e.date) return false;

   912→        const incCount = filtered.filter((e: any) => e.type === 'income').length;
   913→        const expCount = filtered.filter((e: any) => e.type === 'expense').length;
