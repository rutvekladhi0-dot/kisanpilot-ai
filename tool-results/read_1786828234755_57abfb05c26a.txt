    28→export interface AlertItem {
    29→  id: number;
    30→  text: string;
    31→  icon: string;

    35→// ============================================================
    36→// App Context
    37→// ============================================================
    38→interface AppContextType {
    39→  lang: Language;
    40→  t: Translations;
    41→  setLang: (l: Language) => void;
    42→  profile: FarmerProfile | null;
    43→  setProfile: (p: FarmerProfile) => void;
    44→  screen: Screen;
    45→  navigate: (s: Screen) => void;
    46→  alerts: AlertItem[];
    47→  setAlerts: (a: AlertItem[]) => void;
    48→}
    49→
    50→const AppContext = createContext<AppContextType>({} as AppContextType);
    51→
    52→export function useApp() {
    53→  return useContext(AppContext);
    54→}
    55→
    56→const defaultAlerts: AlertItem[] = [
    57→  { id: 1, icon: '🌧', text: 'rainExpected', read: false },
    58→  { id: 2, icon: '💧', text: 'monitorSoil', read: false },
    59→  { id: 3, icon: '🐛', text: 'checkPestSigns', read: false },
    60→  { id: 4, icon: '🧪', text: 'fertilizerReminder', read: false },

    63→// ============================================================
    64→// Main App Component
    65→// ============================================================
    66→export default function KisanPilotApp() {
    67→  const initialized = useRef(false);
    68→  const [lang, setLangState] = useState<Language>(() => {
    69→    if (typeof window !== 'undefined') {
    70→      return (localStorage.getItem('kp_lang') as Language) || 'en';
    71→    }
    72→    return 'en';
    73→  });
    74→  const [profile, setProfileState] = useState<FarmerProfile | null>(() => {
    75→    if (typeof window !== 'undefined') {
    76→      try {
    77→        const saved = localStorage.getItem('kp_profile');
    78→        return saved ? JSON.parse(saved) : null;
    79→      } catch { return null; }
    80→    }
    81→    return null;
    82→  });
    83→  const [screen, setScreen] = useState<Screen>(() => {
    84→    if (typeof window !== 'undefined') {
    85→      const savedLang = localStorage.getItem('kp_lang');
    86→      const savedProfile = localStorage.getItem('kp_profile');
    87→      if (savedLang && savedProfile) return 'dashboard';
    88→      if (savedLang) return 'profile';
    89→    }
    90→    return 'language';
    91→  });
    92→  const [alerts, setAlertsState] = useState<AlertItem[]>(() => {
    93→    if (typeof window !== 'undefined') {
    94→      try {
    95→        const saved = localStorage.getItem('kp_alerts');
    96→        return saved ? JSON.parse(saved) : defaultAlerts;
    97→      } catch { return defaultAlerts; }
    98→    }
    99→    return defaultAlerts;
   100→  });
   101→
   102→  const [mounted, setMounted] = useState(false);
   103→
   104→  useEffect(() => {
   105→    if (!initialized.current) {
   106→      initialized.current = true;
   107→      requestAnimationFrame(() => setMounted(true));
   108→    }
   109→  }, []);
   110→
   111→  const t = getTranslations(lang);
   112→
   113→  const setLang = useCallback((l: Language) => {
   114→    setLangState(l);
   115→    localStorage.setItem('kp_lang', l);
   116→  }, []);
   117→
   118→  const setProfile = useCallback((p: FarmerProfile) => {
   119→    setProfileState(p);
   120→    localStorage.setItem('kp_profile', JSON.stringify(p));
   121→  }, []);
   122→
   123→  const navigate = useCallback((s: Screen) => {
   124→    setScreen(s);
   125→    window.scrollTo(0, 0);
   126→  }, []);
   127→
   128→  const setAlerts = useCallback((a: AlertItem[]) => {
   129→    setAlertsState(a);
   130→    localStorage.setItem('kp_alerts', JSON.stringify(a));
   131→  }, []);
   132→
   133→  if (!mounted) {
   134→    return null;
   135→  }
   136→
   137→  return (
   138→    <AppContext.Provider value={{ lang, t, setLang, profile, setProfile, screen, navigate, alerts, setAlerts }}>
   139→      <div className="min-h-screen bg-gradient-to-br from-green-50 via-white to-emerald-50">
   140→        <AnimatePresence mode="wait">
   141→          {screen === 'language' && <LanguageScreen key="language" />}
   142→          {screen === 'profile' && <ProfileScreen key="profile" />}
   143→          {screen === 'dashboard' && <DashboardScreen key="dashboard" />}
   144→          {screen === 'chatbot' && <ChatbotScreen key="chatbot" />}
   145→          {screen === 'myfarm' && <MyFarmScreen key="myfarm" />}
   146→          {screen === 'cropdoctor' && <CropDoctorScreen key="cropdoctor" />}
   147→          {screen === 'weather' && <WeatherScreen key="weather" />}
   148→          {screen === 'insights' && <InsightsScreen key="insights" />}
   149→          {screen === 'alerts' && <AlertsScreen key="alerts" />}
   150→          {screen === 'settings' && <SettingsScreen key="settings" />}
   151→          {screen === 'voicekhata' && <VoiceKhataScreen key="voicekhata" />}
   152→          {screen === 'farmeconomics' && <FarmEconomicsScreen key="farmeconomics" />}
   153→          {screen === 'bajarbhav' && <BajarBhavScreen key="bajarbhav" />}
   154→          {screen === 'riskradar' && <RiskRadarScreen key="riskradar" />}
   155→          {screen === 'simulator' && <SimulatorScreen key="simulator" />}
   156→          {screen === 'pmfby' && <PMFBYScreen key="pmfby" />}
   157→          {screen === 'farmgoals' && <FarmGoalsScreen key="farmgoals" />}
   158→          {screen === 'nextbestaction' && <NextBestActionScreen key="nextbestaction" />}
   159→          {screen === 'phototracker' && <MonthlyPhotoTrackerScreen key="phototracker" />}
   160→        </AnimatePresence>
   161→      </div>
   162→    </AppContext.Provider>
   163→  );
   164→}

   166→// ============================================================
   167→// Shared Animation Variants
   168→// ============================================================
   169→const pageVariants = {
   170→  initial: { opacity: 0, y: 20 },
   171→  animate: { opacity: 1, y: 0, transition: { duration: 0.3, ease: 'easeOut' } },
   172→  exit: { opacity: 0, y: -20, transition: { duration: 0.2, ease: 'easeIn' } },
   173→};
   174→
   175→function PageWrapper({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   176→  return (
   177→    <motion.div
   178→      variants={pageVariants}
   179→      initial="initial"

   181→      exit="exit"
   182→      className={`min-h-screen ${className}`}
   183→    >
   184→      {children}
   185→    </motion.div>
   186→  );
   187→}

   271→// PROFILE SETUP SCREEN
   272→// ============================================================
   273→function ProfileScreen() {
   274→  const { t, navigate, setProfile } = useApp();
   275→  const [name, setName] = useState('');
   276→  const [village, setVillage] = useState('');
   277→  const [farmSize, setFarmSize] = useState('');
   278→  const [mainCrop, setMainCrop] = useState('');
   279→  const [errors, setErrors] = useState<{ name?: string; village?: string }>({});
   280→
   281→  const farmSizes = [
   282→    { value: '<1', label: t.lessThanOneAcre },
   283→    { value: '1-5', label: t.oneToFiveAcres },
   284→    { value: '5-10', label: t.fiveToTenAcres },
   285→    { value: '>10', label: t.moreThanTenAcres },
   286→  ];
   287→
   288→  const crops = [t.cropWheat, t.cropRice, t.cropCotton, t.cropSoybean, t.cropSugarcane, t.cropTomato, t.cropOnion, t.cropOther];
   289→
   290→  const handleSave = () => {
   291→    const newErrors: { name?: string; village?: string } = {};
   292→    if (!name.trim()) newErrors.name = t.nameRequired;
   293→    if (!village.trim()) newErrors.village = t.villageRequired;
   294→    if (Object.keys(newErrors).length > 0) {
   295→      setErrors(newErrors);
   296→      return;
   297→    }
   298→    setProfile({
   299→      name: name.trim(),
   300→      village: village.trim(),
   301→      farmSize: farmSize || t.lessThanOneAcre,
   302→      mainCrop: mainCrop || t.cropWheat,
   303→    });
   304→    navigate('dashboard');
   305→  };
   306→
   307→  return (
   308→    <PageWrapper className="flex items-center justify-center px-4 py-8">
   309→      <div className="w-full max-w-md">
   310→        <div className="text-center mb-8">
   311→          <div className="w-14 h-14 rounded-2xl bg-gradient-to-br from-green-600 to-emerald-700 flex items-center justify-center shadow-lg mx-auto mb-3">
   312→            <span className="text-2xl">👨‍🌾</span>
   313→          </div>
   314→          <h1 className="text-2xl font-bold text-gray-800">{t.setupProfile}</h1>
   315→          <p className="text-gray-500 mt-1">{t.setupProfileSubtitle}</p>
   316→        </div>
   317→
   318→        <div className="bg-white rounded-3xl shadow-lg p-6 md:p-8 space-y-5">
   319→          {/* Farmer Name */}
   320→          <div>
   321→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.farmerName}</label>
   322→            <input
   323→              type="text"
   324→              value={name}
   325→              onChange={(e) => { setName(e.target.value); setErrors(p => ({ ...p, name: undefined })); }}
   326→              placeholder={t.farmerNamePlaceholder}
   327→              className={`w-full px-4 py-3 rounded-xl border-2 ${errors.name ? 'border-red-400 bg-red-50' : 'border-gray-200 bg-gray-50'} focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800`}
   328→            />
   329→            {errors.name && <p className="text-red-500 text-sm mt-1">{errors.name}</p>}
   330→          </div>
   331→
   332→          {/* Village Name */}
   333→          <div>
   334→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.villageName}</label>
   335→            <input
   336→              type="text"
   337→              value={village}
   338→              onChange={(e) => { setVillage(e.target.value); setErrors(p => ({ ...p, village: undefined })); }}
   339→              placeholder={t.villageNamePlaceholder}
   340→              className={`w-full px-4 py-3 rounded-xl border-2 ${errors.village ? 'border-red-400 bg-red-50' : 'border-gray-200 bg-gray-50'} focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800`}
   341→            />
   342→            {errors.village && <p className="text-red-500 text-sm mt-1">{errors.village}</p>}
   343→          </div>
   344→
   345→          {/* Farm Size */}
   346→          <div>
   347→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.farmSize}</label>
   348→            <div className="grid grid-cols-2 gap-2">
   349→              {farmSizes.map((fs) => (
   350→                <button
   351→                  key={fs.value}
   352→                  onClick={() => setFarmSize(fs.label)}
   353→                  className={`px-3 py-2.5 rounded-xl text-sm font-medium border-2 transition-all ${
   354→                    farmSize === fs.label
   355→                      ? 'border-green-500 bg-green-50 text-green-700'
   356→                      : 'border-gray-200 bg-gray-50 text-gray-600 hover:border-gray-300'
   357→                  }`}
   358→                >
   359→                  {fs.label}
   360→                </button>
   361→              ))}
   362→            </div>
   363→          </div>
   364→
   365→          {/* Main Crop */}
   366→          <div>
   367→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.mainCrop}</label>
   368→            <select
   369→              value={mainCrop}
   370→              onChange={(e) => setMainCrop(e.target.value)}
   371→              className="w-full px-4 py-3 rounded-xl border-2 border-gray-200 bg-gray-50 focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800"
   372→            >
   373→              <option value="">{t.selectCrop}</option>
   374→              {crops.map((c) => (
   375→                <option key={c} value={c}>{c}</option>
   376→              ))}
   377→            </select>
   378→          </div>
   379→
   380→          {/* Buttons */}
   381→          <div className="flex gap-3 pt-2">
   382→            <button
   383→              onClick={() => navigate('language')}
   384→              className="flex-1 py-3 px-4 rounded-xl border-2 border-gray-200 text-gray-600 font-semibold hover:bg-gray-50 transition-all"
   385→            >
   386→              {t.back}
   387→            </button>
   388→            <button
   389→              onClick={handleSave}
   390→              className="flex-1 py-3 px-4 rounded-xl bg-gradient-to-r from-green-600 to-emerald-600 text-white font-semibold shadow-md hover:shadow-lg hover:from-green-700 hover:to-emerald-700 transition-all"
   391→            >
   392→              {t.saveAndContinue}
   393→            </button>
   394→          </div>
   395→        </div>
   396→      </div>
   397→    </PageWrapper>
   398→  );
   399→}

   588→interface ChatMessage {
   589→  id: number;
   590→  role: 'user' | 'bot';
   591→  text: string;
   592→}
   593→
   594→function ChatbotScreen() {
   595→  const { t, lang } = useApp();
   596→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   597→  const [input, setInput] = useState('');
   598→  const [isTyping, setIsTyping] = useState(false);
   599→  const chatEndRef = React.useRef<HTMLDivElement>(null);
   600→
   601→  useEffect(() => {
   602→    chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   603→  }, [messages, isTyping]);
   604→
   605→  const generateResponse = (userMsg: string): string => {
   606→    const lower = userMsg.toLowerCase();
   607→    if (lower.includes('water') || lower.includes('irrigat') || lower.includes('watering') || lower.includes('पानी') || lower.includes('सिंचाई')) {
   608→      return t.waterResponse;
   609→    }
   610→    if (lower.includes('yellow') || lower.includes('पील') || lower.includes('पिवळ')) {
   611→      return t.yellowLeavesResponse;
   612→    }
   613→    if (lower.includes('pest') || lower.includes('insect') || lower.includes('कीट') || lower.includes('bug')) {
   614→      return t.pestResponse;
   615→    }
   616→    if (lower.includes('fertilizer') || lower.includes('nutrient') || lower.includes('उर्वरक') || lower.includes('पोषक') || lower.includes('खत')) {
   617→      return t.fertilizerResponse;
   618→    }
   619→    if (lower.includes('weather') || lower.includes('मौसम') || lower.includes('हवामान')) {
   620→      return t.weatherResponse;
   621→    }
   622→    return t.defaultResponse;
   623→  };
   624→
   625→  const sendMessage = (text?: string) => {
   626→    const msgText = text || input.trim();
   627→    if (!msgText) return;
   628→
   629→    const userMsg: ChatMessage = { id: Date.now(), role: 'user', text: msgText };
   630→    setMessages(prev => [...prev, userMsg]);
   631→    setInput('');
   632→    setIsTyping(true);
   633→
   634→    setTimeout(() => {
   635→      const botReply: ChatMessage = { id: Date.now() + 1, role: 'bot', text: generateResponse(msgText) };
   636→      setIsTyping(false);
   637→      setMessages(prev => [...prev, botReply]);
   638→    }, 800 + Math.random() * 700);
   639→  };
   640→
   641→  const handleKeyDown = (e: React.KeyboardEvent) => {
   642→    if (e.key === 'Enter' && !e.shiftKey) {
   643→      e.preventDefault();
   644→      sendMessage();
   645→    }
   646→  };
   647→
   648→  return (
   649→    <PageWrapper className="flex flex-col">
   650→      {/* Header */}
   651→      <header className="sticky top-0 z-50 bg-white/80 backdrop-blur-xl border-b border-green-100">
   652→        <div className="max-w-2xl mx-auto px-4 py-3 flex items-center gap-3">
   653→          <BackButton target="dashboard" label={t.backToDashboard} />
   654→          <div className="flex-1 flex items-center gap-2 justify-center">
   655→            <span className="text-xl">🤖</span>
   656→            <h1 className="text-lg font-bold text-green-800">{t.aiAssistant}</h1>
   657→          </div>
   658→        </div>
   659→      </header>
   660→
   661→      {/* Messages */}
   662→      <div className="flex-1 max-w-2xl mx-auto w-full px-4 py-4 overflow-y-auto">
   663→        {messages.length === 0 && (
   664→          <div className="text-center py-12">
   665→            <div className="text-5xl mb-4">🤖</div>
   666→            <h3 className="text-xl font-bold text-gray-800 mb-2">{t.aiAssistant}</h3>
   667→            <p className="text-gray-500 text-sm mb-8 max-w-sm mx-auto">{t.defaultResponse}</p>
   668→
   669→            {/* Quick Suggestions */}
   670→            <div className="space-y-2">
   671→              <p className="text-sm font-semibold text-gray-600 mb-3">{t.quickSuggestions}</p>
   672→              {[t.suggestion1, t.suggestion2, t.suggestion3, t.suggestion4].map((s, i) => (
   673→                <motion.button
   674→                  key={i}
   675→                  initial={{ opacity: 0, y: 10 }}
   676→                  animate={{ opacity: 1, y: 0 }}
   677→                  transition={{ delay: i * 0.1 }}
   678→                  onClick={() => sendMessage(s)}
   679→                  className="w-full text-left px-4 py-3 bg-white rounded-xl border border-green-200 text-sm text-green-800 hover:bg-green-50 hover:border-green-300 transition-all shadow-sm"
   680→                >
   681→                  {s}
   682→                </motion.button>
   683→              ))}
   684→            </div>
   685→          </div>
   686→        )}
   687→
   688→        {/* Message List */}
   689→        {messages.map((msg) => (
   690→          <motion.div
   691→            key={msg.id}
   692→            initial={{ opacity: 0, y: 10 }}
   693→            animate={{ opacity: 1, y: 0 }}
   694→            className={`flex mb-4 ${msg.role === 'user' ? 'justify-end' : 'justify-start'}`}
   695→          >
   696→            <div className={`max-w-[85%] rounded-2xl px-4 py-3 text-sm leading-relaxed ${
   697→              msg.role === 'user'
   698→                ? 'bg-gradient-to-r from-green-600 to-emerald-600 text-white rounded-br-md'
   699→                : 'bg-white border border-gray-200 text-gray-800 shadow-sm rounded-bl-md'
   700→            }`}>
   701→              <div className="whitespace-pre-wrap break-words">{msg.text}</div>
   702→            </div>
   703→          </motion.div>
   704→        ))}
   705→
   706→        {isTyping && (
   707→          <div className="flex mb-4 justify-start">
   708→            <div className="bg-white border border-gray-200 rounded-2xl rounded-bl-md px-4 py-3 shadow-sm">
   709→              <div className="flex gap-1.5">
   710→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '0ms' }} />
   711→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '150ms' }} />
   712→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '300ms' }} />
   713→              </div>
   714→            </div>
   715→          </div>
   716→        )}
   717→        <div ref={chatEndRef} />
   718→      </div>
   719→
   720→      {/* Input */}
   721→      <div className="sticky bottom-0 bg-white/80 backdrop-blur-xl border-t border-green-100">
   722→        <div className="max-w-2xl mx-auto px-4 py-3">
   723→          <div className="flex gap-2">
   724→            <input
   725→              type="text"
   726→              value={input}
   727→              onChange={(e) => setInput(e.target.value)}
   728→              onKeyDown={handleKeyDown}
   729→              placeholder={t.typeMessage}
   730→              className="flex-1 px-4 py-3 rounded-xl border-2 border-gray-200 bg-gray-50 focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800 text-sm"
   731→            />
   732→            <button
   733→              onClick={() => sendMessage()}
   734→              disabled={!input.trim() || isTyping}
   735→              className="px-5 py-3 rounded-xl bg-gradient-to-r from-green-600 to-emerald-600 text-white font-semibold shadow-md hover:shadow-lg hover:from-green-700 hover:to-emerald-700 disabled:opacity-50 disabled:cursor-not-allowed transition-all"
   736→            >
   737→              {t.send}
   738→            </button>
   739→          </div>
   740→        </div>
   741→      </div>
   742→    </PageWrapper>
