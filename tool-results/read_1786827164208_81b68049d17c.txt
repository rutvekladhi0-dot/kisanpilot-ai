    63→    63→// Main App Component
    64→    64→// ============================================================
    65→    65→export default function KisanPilotApp() {
    66→    66→  const initialized = useRef(false);
    67→    67→  const [lang, setLangState] = useState<Language>(() => {
    68→    68→    if (typeof window !== 'undefined') {
    69→    69→      return (localStorage.getItem('kp_lang') as Language) || 'en';
    70→    70→    }
    71→    71→    return 'en';
    72→    72→  });
    73→    73→  const [profile, setProfileState] = useState<FarmerProfile | null>(() => {
    74→    74→    if (typeof window !== 'undefined') {
    75→    75→      try {
    76→    76→        const saved = localStorage.getItem('kp_profile');
    77→    77→        return saved ? JSON.parse(saved) : null;
    78→    78→      } catch { return null; }
    79→    79→    }
    80→    80→    return null;
    81→    81→  });
    82→    82→  const [screen, setScreen] = useState<Screen>(() => {
    83→    83→    if (typeof window !== 'undefined') {
    84→    84→      const savedLang = localStorage.getItem('kp_lang');
    85→    85→      const savedProfile = localStorage.getItem('kp_profile');
    86→    86→      if (savedLang && savedProfile) return 'dashboard';
    87→    87→      if (savedLang) return 'profile';
    88→    88→    }
    89→    89→    return 'language';
    90→    90→  });
    91→    91→  const [alerts, setAlertsState] = useState<AlertItem[]>(() => {
    92→    92→    if (typeof window !== 'undefined') {
    93→    93→      try {
    94→    94→        const saved = localStorage.getItem('kp_alerts');
    95→    95→        return saved ? JSON.parse(saved) : defaultAlerts;
    96→    96→      } catch { return defaultAlerts; }
    97→    97→    }
    98→    98→    return defaultAlerts;
    99→    99→  });
   100→   100→
   101→   101→  const [mounted, setMounted] = useState(false);
   102→   102→
   103→   103→  useEffect(() => {
   104→   104→    if (!initialized.current) {
   105→   105→      initialized.current = true;
   106→   106→      requestAnimationFrame(() => setMounted(true));
   107→   107→    }
   108→   108→  }, []);
   109→   109→
   110→   110→  const t = getTranslations(lang);
   111→   111→
   112→   112→  const setLang = useCallback((l: Language) => {
   113→   113→    setLangState(l);
   114→   114→    localStorage.setItem('kp_lang', l);
   115→   115→  }, []);
   116→   116→
   117→   117→  const setProfile = useCallback((p: FarmerProfile) => {
   118→   118→    setProfileState(p);
   119→   119→    localStorage.setItem('kp_profile', JSON.stringify(p));
   120→   120→  }, []);
   121→   121→
   122→   122→  const navigate = useCallback((s: Screen) => {
   123→   123→    setScreen(s);
   124→   124→    window.scrollTo(0, 0);
   125→   125→  }, []);
   126→   126→
   127→   127→  const setAlerts = useCallback((a: AlertItem[]) => {
   128→   128→    setAlertsState(a);
   129→   129→    localStorage.setItem('kp_alerts', JSON.stringify(a));
   130→   130→  }, []);
   131→   131→
   132→   132→  if (!mounted) {
   133→   133→    return null;
   134→   134→  }
   135→   135→
   136→   136→  return (
   137→   137→    <AppContext.Provider value={{ lang, t, setLang, profile, setProfile, screen, navigate, alerts, setAlerts }}>
   138→   138→      <div className="min-h-screen bg-gradient-to-br from-green-50 via-white to-emerald-50">
   139→   139→        <AnimatePresence mode="wait">
   140→   140→          {screen === 'language' && <LanguageScreen key="language" />}
   141→   141→          {screen === 'profile' && <ProfileScreen key="profile" />}
   142→   142→          {screen === 'dashboard' && <DashboardScreen key="dashboard" />}
   143→   143→          {screen === 'chatbot' && <ChatbotScreen key="chatbot" />}
   144→   144→          {screen === 'myfarm' && <MyFarmScreen key="myfarm" />}
   145→   145→          {screen === 'cropdoctor' && <CropDoctorScreen key="cropdoctor" />}
   146→   146→          {screen === 'weather' && <WeatherScreen key="weather" />}
   147→   147→          {screen === 'insights' && <InsightsScreen key="insights" />}
   148→   148→          {screen === 'alerts' && <AlertsScreen key="alerts" />}
   149→   149→          {screen === 'settings' && <SettingsScreen key="settings" />}
   150→   150→          {screen === 'voicekhata' && <VoiceKhataScreen key="voicekhata" />}
   151→   151→          {screen === 'farmeconomics' && <FarmEconomicsScreen key="farmeconomics" />}
   152→   152→          {screen === 'bajarbhav' && <BajarBhavScreen key="bajarbhav" />}
   153→   153→          {screen === 'riskradar' && <RiskRadarScreen key="riskradar" />}
   154→   154→          {screen === 'simulator' && <SimulatorScreen key="simulator" />}
   155→   155→          {screen === 'pmfby' && <PMFBYScreen key="pmfby" />}
   156→   156→          {screen === 'farmgoals' && <FarmGoalsScreen key="farmgoals" />}
   157→   157→          {screen === 'nextbestaction' && <NextBestActionScreen key="nextbestaction" />}
   158→   158→        </AnimatePresence>
   159→   159→      </div>
   160→   160→    </AppContext.Provider>

   585→   585→interface ChatMessage {
   586→   586→  id: number;
   587→   587→  role: 'user' | 'bot';
   588→   588→  text: string;
   589→   589→}
   590→   590→
   591→   591→function ChatbotScreen() {
   592→   592→  const { t, lang } = useApp();
   593→   593→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   594→   594→  const [input, setInput] = useState('');
   595→   595→  const [isTyping, setIsTyping] = useState(false);
   596→   596→  const chatEndRef = React.useRef<HTMLDivElement>(null);
   597→   597→
   598→   598→  useEffect(() => {
   599→   599→    chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   600→   600→  }, [messages, isTyping]);
   601→   601→
   602→   602→  const generateResponse = (userMsg: string): string => {
   603→   603→    const lower = userMsg.toLowerCase();
   604→   604→    if (lower.includes('water') || lower.includes('irrigat') || lower.includes('watering') || lower.includes('पानी') || lower.includes('सिंचाई')) {
   605→   605→      return t.waterResponse;
   606→   606→    }
   607→   607→    if (lower.includes('yellow') || lower.includes('पील') || lower.includes('पिवळ')) {
   608→   608→      return t.yellowLeavesResponse;
   609→   609→    }
   610→   610→    if (lower.includes('pest') || lower.includes('insect') || lower.includes('कीट') || lower.includes('bug')) {
   611→   611→      return t.pestResponse;
   612→   612→    }
   613→   613→    if (lower.includes('fertilizer') || lower.includes('nutrient') || lower.includes('उर्वरक') || lower.includes('पोषक') || lower.includes('खत')) {
   614→   614→      return t.fertilizerResponse;
   615→   615→    }
   616→   616→    if (lower.includes('weather') || lower.includes('मौसम') || lower.includes('हवामान')) {
   617→   617→      return t.weatherResponse;
   618→   618→    }
   619→   619→    return t.defaultResponse;
   620→   620→  };
   621→   621→
   622→   622→  const sendMessage = (text?: string) => {
   623→   623→    const msgText = text || input.trim();
   624→   624→    if (!msgText) return;
   625→   625→
   626→   626→    const userMsg: ChatMessage = { id: Date.now(), role: 'user', text: msgText };
   627→   627→    setMessages(prev => [...prev, userMsg]);
   628→   628→    setInput('');
   629→   629→    setIsTyping(true);
   630→   630→
   631→   631→    setTimeout(() => {
   632→   632→      const botReply: ChatMessage = { id: Date.now() + 1, role: 'bot', text: generateResponse(msgText) };
   633→   633→      setIsTyping(false);
   634→   634→      setMessages(prev => [...prev, botReply]);
   635→   635→    }, 800 + Math.random() * 700);
   636→   636→  };
   637→   637→
   638→   638→  const handleKeyDown = (e: React.KeyboardEvent) => {
   639→   639→    if (e.key === 'Enter' && !e.shiftKey) {
   640→   640→      e.preventDefault();
   641→   641→      sendMessage();
   642→   642→    }
   643→   643→  };
   644→   644→
   645→   645→  return (
   646→   646→    <PageWrapper className="flex flex-col">
   647→   647→      {/* Header */}
   648→   648→      <header className="sticky top-0 z-50 bg-white/80 backdrop-blur-xl border-b border-green-100">
   649→   649→        <div className="max-w-2xl mx-auto px-4 py-3 flex items-center gap-3">
   650→   650→          <BackButton target="dashboard" label={t.backToDashboard} />
   651→   651→          <div className="flex-1 flex items-center gap-2 justify-center">
   652→   652→            <span className="text-xl">🤖</span>
   653→   653→            <h1 className="text-lg font-bold text-green-800">{t.aiAssistant}</h1>
   654→   654→          </div>
   655→   655→        </div>
   656→   656→      </header>
   657→   657→
   658→   658→      {/* Messages */}
   659→   659→      <div className="flex-1 max-w-2xl mx-auto w-full px-4 py-4 overflow-y-auto">
   660→   660→        {messages.length === 0 && (
   661→   661→          <div className="text-center py-12">
   662→   662→            <div className="text-5xl mb-4">🤖</div>
   663→   663→            <h3 className="text-xl font-bold text-gray-800 mb-2">{t.aiAssistant}</h3>
   664→   664→            <p className="text-gray-500 text-sm mb-8 max-w-sm mx-auto">{t.defaultResponse}</p>
   665→   665→
   666→   666→            {/* Quick Suggestions */}
   667→   667→            <div className="space-y-2">
   668→   668→              <p className="text-sm font-semibold text-gray-600 mb-3">{t.quickSuggestions}</p>
   669→   669→              {[t.suggestion1, t.suggestion2, t.suggestion3, t.suggestion4].map((s, i) => (
   670→   670→                <motion.button
   671→   671→                  key={i}
   672→   672→                  initial={{ opacity: 0, y: 10 }}
   673→   673→                  animate={{ opacity: 1, y: 0 }}
   674→   674→                  transition={{ delay: i * 0.1 }}
   675→   675→                  onClick={() => sendMessage(s)}
   676→   676→                  className="w-full text-left px-4 py-3 bg-white rounded-xl border border-green-200 text-sm text-green-800 hover:bg-green-50 hover:border-green-300 transition-all shadow-sm"
   677→   677→                >
   678→   678→                  {s}
   679→   679→                </motion.button>
   680→   680→              ))}
   681→   681→            </div>
   682→   682→          </div>
   683→   683→        )}
   684→   684→
   685→   685→        {/* Message List */}
   686→   686→        {messages.map((msg) => (
   687→   687→          <motion.div
   688→   688→            key={msg.id}
   689→   689→            initial={{ opacity: 0, y: 10 }}
   690→   690→            animate={{ opacity: 1, y: 0 }}
   691→   691→            className={`flex mb-4 ${msg.role === 'user' ? 'justify-end' : 'justify-start'}`}
   692→   692→          >
   693→   693→            <div className={`max-w-[85%] rounded-2xl px-4 py-3 text-sm leading-relaxed ${
   694→   694→              msg.role === 'user'
   695→   695→                ? 'bg-gradient-to-r from-green-600 to-emerald-600 text-white rounded-br-md'
   696→   696→                : 'bg-white border border-gray-200 text-gray-800 shadow-sm rounded-bl-md'
   697→   697→            }`}>
   698→   698→              <div className="whitespace-pre-wrap break-words">{msg.text}</div>
   699→   699→            </div>
   700→   700→          </motion.div>
   701→   701→        ))}
   702→   702→
   703→   703→        {isTyping && (
   704→   704→          <div className="flex mb-4 justify-start">
   705→   705→            <div className="bg-white border border-gray-200 rounded-2xl rounded-bl-md px-4 py-3 shadow-sm">
   706→   706→              <div className="flex gap-1.5">
   707→   707→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '0ms' }} />
   708→   708→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '150ms' }} />
   709→   709→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '300ms' }} />
   710→   710→              </div>
   711→   711→            </div>
   712→   712→          </div>
   713→   713→        )}
   714→   714→        <div ref={chatEndRef} />
   715→   715→      </div>
   716→   716→
   717→   717→      {/* Input */}
   718→   718→      <div className="sticky bottom-0 bg-white/80 backdrop-blur-xl border-t border-green-100">
   719→   719→        <div className="max-w-2xl mx-auto px-4 py-3">
   720→   720→          <div className="flex gap-2">
   721→   721→            <input
   722→   722→              type="text"
   723→   723→              value={input}
   724→   724→              onChange={(e) => setInput(e.target.value)}
   725→   725→              onKeyDown={handleKeyDown}
   726→   726→              placeholder={t.typeMessage}
   727→   727→              className="flex-1 px-4 py-3 rounded-xl border-2 border-gray-200 bg-gray-50 focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800 text-sm"
   728→   728→            />
   729→   729→            <button
   730→   730→              onClick={() => sendMessage()}
   731→   731→              disabled={!input.trim() || isTyping}
   732→   732→              className="px-5 py-3 rounded-xl bg-gradient-to-r from-green-600 to-emerald-600 text-white font-semibold shadow-md hover:shadow-lg hover:from-green-700 hover:to-emerald-700 disabled:opacity-50 disabled:cursor-not-allowed transition-all"
   733→   733→            >
   734→   734→              {t.send}
   735→   735→            </button>
   736→   736→          </div>
   737→   737→        </div>
   738→   738→      </div>
   739→   739→    </PageWrapper>
   740→   740→  );
   741→   741→}
