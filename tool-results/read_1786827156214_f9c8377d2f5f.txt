    27→export interface AlertItem {
    28→  id: number;
    29→  text: string;
    30→  icon: string;

    34→// ============================================================
    35→// App Context
    36→// ============================================================
    37→interface AppContextType {
    38→  lang: Language;
    39→  t: Translations;
    40→  setLang: (l: Language) => void;
    41→  profile: FarmerProfile | null;
    42→  setProfile: (p: FarmerProfile) => void;
    43→  screen: Screen;
    44→  navigate: (s: Screen) => void;
    45→  alerts: AlertItem[];
    46→  setAlerts: (a: AlertItem[]) => void;
    47→}
    48→
    49→const AppContext = createContext<AppContextType>({} as AppContextType);
    50→
    51→export function useApp() {
    52→  return useContext(AppContext);
    53→}
    54→
    55→const defaultAlerts: AlertItem[] = [
    56→  { id: 1, icon: '🌧', text: 'rainExpected', read: false },
    57→  { id: 2, icon: '💧', text: 'monitorSoil', read: false },
    58→  { id: 3, icon: '🐛', text: 'checkPestSigns', read: false },
    59→  { id: 4, icon: '🧪', text: 'fertilizerReminder', read: false },

    62→// ============================================================
    63→// Main App Component
    64→// ============================================================
    65→export default function KisanPilotApp() {
    66→  const initialized = useRef(false);
    67→  const [lang, setLangState] = useState<Language>(() => {
    68→    if (typeof window !== 'undefined') {
    69→      return (localStorage.getItem('kp_lang') as Language) || 'en';
    70→    }
    71→    return 'en';
    72→  });
    73→  const [profile, setProfileState] = useState<FarmerProfile | null>(() => {
    74→    if (typeof window !== 'undefined') {
    75→      try {
    76→        const saved = localStorage.getItem('kp_profile');
    77→        return saved ? JSON.parse(saved) : null;
    78→      } catch { return null; }
    79→    }
    80→    return null;
    81→  });
    82→  const [screen, setScreen] = useState<Screen>(() => {
    83→    if (typeof window !== 'undefined') {
    84→      const savedLang = localStorage.getItem('kp_lang');
    85→      const savedProfile = localStorage.getItem('kp_profile');
    86→      if (savedLang && savedProfile) return 'dashboard';
    87→      if (savedLang) return 'profile';
    88→    }
    89→    return 'language';
    90→  });
    91→  const [alerts, setAlertsState] = useState<AlertItem[]>(() => {
    92→    if (typeof window !== 'undefined') {
    93→      try {
    94→        const saved = localStorage.getItem('kp_alerts');
    95→        return saved ? JSON.parse(saved) : defaultAlerts;
    96→      } catch { return defaultAlerts; }
    97→    }
    98→    return defaultAlerts;
    99→  });
   100→
   101→  const [mounted, setMounted] = useState(false);
   102→
   103→  useEffect(() => {
   104→    if (!initialized.current) {
   105→      initialized.current = true;
   106→      requestAnimationFrame(() => setMounted(true));
   107→    }
   108→  }, []);
   109→
   110→  const t = getTranslations(lang);
   111→
   112→  const setLang = useCallback((l: Language) => {
   113→    setLangState(l);
   114→    localStorage.setItem('kp_lang', l);
   115→  }, []);
   116→
   117→  const setProfile = useCallback((p: FarmerProfile) => {
   118→    setProfileState(p);
   119→    localStorage.setItem('kp_profile', JSON.stringify(p));
   120→  }, []);
   121→
   122→  const navigate = useCallback((s: Screen) => {
   123→    setScreen(s);
   124→    window.scrollTo(0, 0);
   125→  }, []);
   126→
   127→  const setAlerts = useCallback((a: AlertItem[]) => {
   128→    setAlertsState(a);
   129→    localStorage.setItem('kp_alerts', JSON.stringify(a));
   130→  }, []);
   131→
   132→  if (!mounted) {
   133→    return null;
   134→  }
   135→
   136→  return (
   137→    <AppContext.Provider value={{ lang, t, setLang, profile, setProfile, screen, navigate, alerts, setAlerts }}>
   138→      <div className="min-h-screen bg-gradient-to-br from-green-50 via-white to-emerald-50">
   139→        <AnimatePresence mode="wait">
   140→          {screen === 'language' && <LanguageScreen key="language" />}
   141→          {screen === 'profile' && <ProfileScreen key="profile" />}
   142→          {screen === 'dashboard' && <DashboardScreen key="dashboard" />}
   143→          {screen === 'chatbot' && <ChatbotScreen key="chatbot" />}
   144→          {screen === 'myfarm' && <MyFarmScreen key="myfarm" />}
   145→          {screen === 'cropdoctor' && <CropDoctorScreen key="cropdoctor" />}
   146→          {screen === 'weather' && <WeatherScreen key="weather" />}
   147→          {screen === 'insights' && <InsightsScreen key="insights" />}
   148→          {screen === 'alerts' && <AlertsScreen key="alerts" />}
   149→          {screen === 'settings' && <SettingsScreen key="settings" />}
   150→          {screen === 'voicekhata' && <VoiceKhataScreen key="voicekhata" />}
   151→          {screen === 'farmeconomics' && <FarmEconomicsScreen key="farmeconomics" />}
   152→          {screen === 'bajarbhav' && <BajarBhavScreen key="bajarbhav" />}
   153→          {screen === 'riskradar' && <RiskRadarScreen key="riskradar" />}
   154→          {screen === 'simulator' && <SimulatorScreen key="simulator" />}
   155→          {screen === 'pmfby' && <PMFBYScreen key="pmfby" />}
   156→          {screen === 'farmgoals' && <FarmGoalsScreen key="farmgoals" />}
   157→          {screen === 'nextbestaction' && <NextBestActionScreen key="nextbestaction" />}
   158→        </AnimatePresence>
   159→      </div>
   160→    </AppContext.Provider>
   161→  );
   162→}
   163→
   164→// ============================================================
   165→// Shared Animation Variants
   166→// ============================================================
   167→const pageVariants = {
   168→  initial: { opacity: 0, y: 20 },
   169→  animate: { opacity: 1, y: 0, transition: { duration: 0.3, ease: 'easeOut' } },
   170→  exit: { opacity: 0, y: -20, transition: { duration: 0.2, ease: 'easeIn' } },
   171→};
   172→
   173→function PageWrapper({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   174→  return (
   175→    <motion.div
   176→      variants={pageVariants}
   177→      initial="initial"
   178→      animate="animate"
   179→      exit="exit"
   180→      className={`min-h-screen ${className}`}
   181→    >
   182→      {children}
   183→    </motion.div>
   184→  );
   185→}

   269→// PROFILE SETUP SCREEN
   270→// ============================================================
   271→function ProfileScreen() {
   272→  const { t, navigate, setProfile } = useApp();
   273→  const [name, setName] = useState('');
   274→  const [village, setVillage] = useState('');
   275→  const [farmSize, setFarmSize] = useState('');
   276→  const [mainCrop, setMainCrop] = useState('');
   277→  const [errors, setErrors] = useState<{ name?: string; village?: string }>({});
   278→
   279→  const farmSizes = [
   280→    { value: '<1', label: t.lessThanOneAcre },
   281→    { value: '1-5', label: t.oneToFiveAcres },
   282→    { value: '5-10', label: t.fiveToTenAcres },
   283→    { value: '>10', label: t.moreThanTenAcres },
   284→  ];
   285→
   286→  const crops = [t.cropWheat, t.cropRice, t.cropCotton, t.cropSoybean, t.cropSugarcane, t.cropTomato, t.cropOnion, t.cropOther];
   287→
   288→  const handleSave = () => {
   289→    const newErrors: { name?: string; village?: string } = {};
   290→    if (!name.trim()) newErrors.name = t.nameRequired;
   291→    if (!village.trim()) newErrors.village = t.villageRequired;
   292→    if (Object.keys(newErrors).length > 0) {
   293→      setErrors(newErrors);
   294→      return;
   295→    }
   296→    setProfile({
   297→      name: name.trim(),
   298→      village: village.trim(),
   299→      farmSize: farmSize || t.lessThanOneAcre,
   300→      mainCrop: mainCrop || t.cropWheat,
   301→    });
   302→    navigate('dashboard');
   303→  };
   304→
   305→  return (
   306→    <PageWrapper className="flex items-center justify-center px-4 py-8">
   307→      <div className="w-full max-w-md">
   308→        <div className="text-center mb-8">
   309→          <div className="w-14 h-14 rounded-2xl bg-gradient-to-br from-green-600 to-emerald-700 flex items-center justify-center shadow-lg mx-auto mb-3">
   310→            <span className="text-2xl">👨‍🌾</span>
   311→          </div>
   312→          <h1 className="text-2xl font-bold text-gray-800">{t.setupProfile}</h1>
   313→          <p className="text-gray-500 mt-1">{t.setupProfileSubtitle}</p>
   314→        </div>
   315→
   316→        <div className="bg-white rounded-3xl shadow-lg p-6 md:p-8 space-y-5">
   317→          {/* Farmer Name */}
   318→          <div>
   319→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.farmerName}</label>
   320→            <input
   321→              type="text"
   322→              value={name}
   323→              onChange={(e) => { setName(e.target.value); setErrors(p => ({ ...p, name: undefined })); }}
   324→              placeholder={t.farmerNamePlaceholder}
   325→              className={`w-full px-4 py-3 rounded-xl border-2 ${errors.name ? 'border-red-400 bg-red-50' : 'border-gray-200 bg-gray-50'} focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800`}
   326→            />
   327→            {errors.name && <p className="text-red-500 text-sm mt-1">{errors.name}</p>}
   328→          </div>
   329→
   330→          {/* Village Name */}
   331→          <div>
   332→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.villageName}</label>
   333→            <input
   334→              type="text"
   335→              value={village}
   336→              onChange={(e) => { setVillage(e.target.value); setErrors(p => ({ ...p, village: undefined })); }}
   337→              placeholder={t.villageNamePlaceholder}
   338→              className={`w-full px-4 py-3 rounded-xl border-2 ${errors.village ? 'border-red-400 bg-red-50' : 'border-gray-200 bg-gray-50'} focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800`}
   339→            />
   340→            {errors.village && <p className="text-red-500 text-sm mt-1">{errors.village}</p>}
   341→          </div>
   342→
   343→          {/* Farm Size */}
   344→          <div>
   345→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.farmSize}</label>
   346→            <div className="grid grid-cols-2 gap-2">
   347→              {farmSizes.map((fs) => (
   348→                <button
   349→                  key={fs.value}
   350→                  onClick={() => setFarmSize(fs.label)}
   351→                  className={`px-3 py-2.5 rounded-xl text-sm font-medium border-2 transition-all ${
   352→                    farmSize === fs.label
   353→                      ? 'border-green-500 bg-green-50 text-green-700'
   354→                      : 'border-gray-200 bg-gray-50 text-gray-600 hover:border-gray-300'
   355→                  }`}
   356→                >
   357→                  {fs.label}
   358→                </button>
   359→              ))}
   360→            </div>
   361→          </div>
   362→
   363→          {/* Main Crop */}
   364→          <div>
   365→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.mainCrop}</label>
   366→            <select
   367→              value={mainCrop}
   368→              onChange={(e) => setMainCrop(e.target.value)}
   369→              className="w-full px-4 py-3 rounded-xl border-2 border-gray-200 bg-gray-50 focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800"
   370→            >
   371→              <option value="">{t.selectCrop}</option>
   372→              {crops.map((c) => (
   373→                <option key={c} value={c}>{c}</option>
   374→              ))}
   375→            </select>
   376→          </div>
   377→
   378→          {/* Buttons */}
   379→          <div className="flex gap-3 pt-2">
   380→            <button
   381→              onClick={() => navigate('language')}
   382→              className="flex-1 py-3 px-4 rounded-xl border-2 border-gray-200 text-gray-600 font-semibold hover:bg-gray-50 transition-all"
   383→            >
   384→              {t.back}
   385→            </button>
   386→            <button
   387→              onClick={handleSave}
   388→              className="flex-1 py-3 px-4 rounded-xl bg-gradient-to-r from-green-600 to-emerald-600 text-white font-semibold shadow-md hover:shadow-lg hover:from-green-700 hover:to-emerald-700 transition-all"
   389→            >
   390→              {t.saveAndContinue}
   391→            </button>
   392→          </div>
   393→        </div>
   394→      </div>
   395→    </PageWrapper>
   396→  );
   397→}

   585→interface ChatMessage {
   586→  id: number;
   587→  role: 'user' | 'bot';
   588→  text: string;
   589→}
   590→
   591→function ChatbotScreen() {
   592→  const { t, lang } = useApp();
   593→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   594→  const [input, setInput] = useState('');
   595→  const [isTyping, setIsTyping] = useState(false);
   596→  const chatEndRef = React.useRef<HTMLDivElement>(null);
   597→
   598→  useEffect(() => {
   599→    chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   600→  }, [messages, isTyping]);
   601→
   602→  const generateResponse = (userMsg: string): string => {
   603→    const lower = userMsg.toLowerCase();
   604→    if (lower.includes('water') || lower.includes('irrigat') || lower.includes('watering') || lower.includes('पानी') || lower.includes('सिंचाई')) {
   605→      return t.waterResponse;
   606→    }
   607→    if (lower.includes('yellow') || lower.includes('पील') || lower.includes('पिवळ')) {
   608→      return t.yellowLeavesResponse;
   609→    }
   610→    if (lower.includes('pest') || lower.includes('insect') || lower.includes('कीट') || lower.includes('bug')) {
   611→      return t.pestResponse;
   612→    }
   613→    if (lower.includes('fertilizer') || lower.includes('nutrient') || lower.includes('उर्वरक') || lower.includes('पोषक') || lower.includes('खत')) {
   614→      return t.fertilizerResponse;
   615→    }
   616→    if (lower.includes('weather') || lower.includes('मौसम') || lower.includes('हवामान')) {
   617→      return t.weatherResponse;
   618→    }
   619→    return t.defaultResponse;
   620→  };
   621→
   622→  const sendMessage = (text?: string) => {
   623→    const msgText = text || input.trim();
   624→    if (!msgText) return;
   625→
   626→    const userMsg: ChatMessage = { id: Date.now(), role: 'user', text: msgText };
   627→    setMessages(prev => [...prev, userMsg]);
   628→    setInput('');
   629→    setIsTyping(true);
   630→
   631→    setTimeout(() => {
   632→      const botReply: ChatMessage = { id: Date.now() + 1, role: 'bot', text: generateResponse(msgText) };
   633→      setIsTyping(false);
   634→      setMessages(prev => [...prev, botReply]);
   635→    }, 800 + Math.random() * 700);
   636→  };
   637→
   638→  const handleKeyDown = (e: React.KeyboardEvent) => {
   639→    if (e.key === 'Enter' && !e.shiftKey) {
   640→      e.preventDefault();
   641→      sendMessage();
   642→    }
   643→  };
   644→
   645→  return (
   646→    <PageWrapper className="flex flex-col">
   647→      {/* Header */}
   648→      <header className="sticky top-0 z-50 bg-white/80 backdrop-blur-xl border-b border-green-100">
   649→        <div className="max-w-2xl mx-auto px-4 py-3 flex items-center gap-3">
   650→          <BackButton target="dashboard" label={t.backToDashboard} />
   651→          <div className="flex-1 flex items-center gap-2 justify-center">
   652→            <span className="text-xl">🤖</span>
   653→            <h1 className="text-lg font-bold text-green-800">{t.aiAssistant}</h1>
   654→          </div>
   655→        </div>
   656→      </header>
   657→
   658→      {/* Messages */}
   659→      <div className="flex-1 max-w-2xl mx-auto w-full px-4 py-4 overflow-y-auto">
   660→        {messages.length === 0 && (
   661→          <div className="text-center py-12">
   662→            <div className="text-5xl mb-4">🤖</div>
   663→            <h3 className="text-xl font-bold text-gray-800 mb-2">{t.aiAssistant}</h3>
   664→            <p className="text-gray-500 text-sm mb-8 max-w-sm mx-auto">{t.defaultResponse}</p>
   665→
   666→            {/* Quick Suggestions */}
   667→            <div className="space-y-2">
   668→              <p className="text-sm font-semibold text-gray-600 mb-3">{t.quickSuggestions}</p>
   669→              {[t.suggestion1, t.suggestion2, t.suggestion3, t.suggestion4].map((s, i) => (
   670→                <motion.button
   671→                  key={i}
   672→                  initial={{ opacity: 0, y: 10 }}
   673→                  animate={{ opacity: 1, y: 0 }}
   674→                  transition={{ delay: i * 0.1 }}
   675→                  onClick={() => sendMessage(s)}
   676→                  className="w-full text-left px-4 py-3 bg-white rounded-xl border border-green-200 text-sm text-green-800 hover:bg-green-50 hover:border-green-300 transition-all shadow-sm"
   677→                >
   678→                  {s}
   679→                </motion.button>
   680→              ))}
   681→            </div>
   682→          </div>
   683→        )}
   684→
   685→        {/* Message List */}
   686→        {messages.map((msg) => (
   687→          <motion.div
   688→            key={msg.id}
   689→            initial={{ opacity: 0, y: 10 }}
   690→            animate={{ opacity: 1, y: 0 }}
   691→            className={`flex mb-4 ${msg.role === 'user' ? 'justify-end' : 'justify-start'}`}
   692→          >
   693→            <div className={`max-w-[85%] rounded-2xl px-4 py-3 text-sm leading-relaxed ${
   694→              msg.role === 'user'
   695→                ? 'bg-gradient-to-r from-green-600 to-emerald-600 text-white rounded-br-md'
   696→                : 'bg-white border border-gray-200 text-gray-800 shadow-sm rounded-bl-md'
   697→            }`}>
   698→              <div className="whitespace-pre-wrap break-words">{msg.text}</div>
   699→            </div>
   700→          </motion.div>
   701→        ))}
   702→
   703→        {isTyping && (
   704→          <div className="flex mb-4 justify-start">
   705→            <div className="bg-white border border-gray-200 rounded-2xl rounded-bl-md px-4 py-3 shadow-sm">
   706→              <div className="flex gap-1.5">
   707→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '0ms' }} />
   708→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '150ms' }} />
   709→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '300ms' }} />
   710→              </div>
   711→            </div>
   712→          </div>
   713→        )}
   714→        <div ref={chatEndRef} />
   715→      </div>
   716→
   717→      {/* Input */}
   718→      <div className="sticky bottom-0 bg-white/80 backdrop-blur-xl border-t border-green-100">
   719→        <div className="max-w-2xl mx-auto px-4 py-3">
   720→          <div className="flex gap-2">
   721→            <input
   722→              type="text"
   723→              value={input}
   724→              onChange={(e) => setInput(e.target.value)}
   725→              onKeyDown={handleKeyDown}
   726→              placeholder={t.typeMessage}
   727→              className="flex-1 px-4 py-3 rounded-xl border-2 border-gray-200 bg-gray-50 focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800 text-sm"
   728→            />
   729→            <button
   730→              onClick={() => sendMessage()}
   731→              disabled={!input.trim() || isTyping}
   732→              className="px-5 py-3 rounded-xl bg-gradient-to-r from-green-600 to-emerald-600 text-white font-semibold shadow-md hover:shadow-lg hover:from-green-700 hover:to-emerald-700 disabled:opacity-50 disabled:cursor-not-allowed transition-all"
   733→            >
   734→              {t.send}
   735→            </button>
   736→          </div>
   737→        </div>
   738→      </div>
   739→    </PageWrapper>
   740→  );
   741→}
