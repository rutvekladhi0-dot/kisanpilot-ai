     1→'use client';
     2→
     3→import React, { useState, useEffect, useCallback, createContext, useContext, useRef } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import { getTranslations, getLanguageLabel, type Language, type Translations } from '@/lib/i18n';
     6→
     7→// ============================================================
     8→// Types
     9→// ============================================================
    10→export type Screen = 'language' | 'profile' | 'dashboard' | 'chatbot' | 'myfarm' | 'cropdoctor' | 'weather' | 'insights' | 'alerts' | 'settings';
    11→
    12→export interface FarmerProfile {
    13→  name: string;
    14→  village: string;
    15→  farmSize: string;
    16→  mainCrop: string;
    17→}
    18→
    19→export interface AlertItem {
    20→  id: number;
    21→  text: string;
    22→  icon: string;
    23→  read: boolean;
    24→}
    25→
    26→// ============================================================
    27→// App Context
    28→// ============================================================
    29→interface AppContextType {
    30→  lang: Language;
    31→  t: Translations;
    32→  setLang: (l: Language) => void;
    33→  profile: FarmerProfile | null;
    34→  setProfile: (p: FarmerProfile) => void;
    35→  screen: Screen;
    36→  navigate: (s: Screen) => void;
    37→  alerts: AlertItem[];
    38→  setAlerts: (a: AlertItem[]) => void;
    39→}
    40→
    41→const AppContext = createContext<AppContextType>({} as AppContextType);
    42→
    43→export function useApp() {
    44→  return useContext(AppContext);
    45→}
    46→
    47→const defaultAlerts: AlertItem[] = [
    48→  { id: 1, icon: '🌧', text: 'rainExpected', read: false },
    49→  { id: 2, icon: '💧', text: 'monitorSoil', read: false },
    50→  { id: 3, icon: '🐛', text: 'checkPestSigns', read: false },
    51→  { id: 4, icon: '🧪', text: 'fertilizerReminder', read: false },
    52→];
    53→
    54→// ============================================================
    55→// Main App Component
    56→// ============================================================
    57→export default function KisanPilotApp() {
    58→  const initialized = useRef(false);
    59→  const [lang, setLangState] = useState<Language>(() => {
    60→    if (typeof window !== 'undefined') {
    61→      return (localStorage.getItem('kp_lang') as Language) || 'en';
    62→    }
    63→    return 'en';
    64→  });
    65→  const [profile, setProfileState] = useState<FarmerProfile | null>(() => {
    66→    if (typeof window !== 'undefined') {
    67→      try {
    68→        const saved = localStorage.getItem('kp_profile');
    69→        return saved ? JSON.parse(saved) : null;
    70→      } catch { return null; }
    71→    }
    72→    return null;
    73→  });
    74→  const [screen, setScreen] = useState<Screen>(() => {
    75→    if (typeof window !== 'undefined') {
    76→      const savedLang = localStorage.getItem('kp_lang');
    77→      const savedProfile = localStorage.getItem('kp_profile');
    78→      if (savedLang && savedProfile) return 'dashboard';
    79→      if (savedLang) return 'profile';
    80→    }
    81→    return 'language';
    82→  });
    83→  const [alerts, setAlertsState] = useState<AlertItem[]>(() => {
    84→    if (typeof window !== 'undefined') {
    85→      try {
    86→        const saved = localStorage.getItem('kp_alerts');
    87→        return saved ? JSON.parse(saved) : defaultAlerts;
    88→      } catch { return defaultAlerts; }
    89→    }
    90→    return defaultAlerts;
    91→  });
    92→
    93→  const [mounted, setMounted] = useState(false);
    94→
    95→  useEffect(() => {
    96→    if (!initialized.current) {
    97→      initialized.current = true;
    98→      requestAnimationFrame(() => setMounted(true));
    99→    }
   100→  }, []);
   101→
   102→  const t = getTranslations(lang);
   103→
   104→  const setLang = useCallback((l: Language) => {
   105→    setLangState(l);
   106→    localStorage.setItem('kp_lang', l);
   107→  }, []);
   108→
   109→  const setProfile = useCallback((p: FarmerProfile) => {
   110→    setProfileState(p);
   111→    localStorage.setItem('kp_profile', JSON.stringify(p));
   112→  }, []);
   113→
   114→  const navigate = useCallback((s: Screen) => {
   115→    setScreen(s);
   116→    window.scrollTo(0, 0);
   117→  }, []);
   118→
   119→  const setAlerts = useCallback((a: AlertItem[]) => {
   120→    setAlertsState(a);
   121→    localStorage.setItem('kp_alerts', JSON.stringify(a));
   122→  }, []);
   123→
   124→  if (!mounted) {
   125→    return null;
   126→  }
   127→
   128→  return (
   129→    <AppContext.Provider value={{ lang, t, setLang, profile, setProfile, screen, navigate, alerts, setAlerts }}>
   130→      <div className="min-h-screen bg-gradient-to-br from-green-50 via-white to-emerald-50">
   131→        <AnimatePresence mode="wait">
   132→          {screen === 'language' && <LanguageScreen key="language" />}
   133→          {screen === 'profile' && <ProfileScreen key="profile" />}
   134→          {screen === 'dashboard' && <DashboardScreen key="dashboard" />}
   135→          {screen === 'chatbot' && <ChatbotScreen key="chatbot" />}
   136→          {screen === 'myfarm' && <MyFarmScreen key="myfarm" />}
   137→          {screen === 'cropdoctor' && <CropDoctorScreen key="cropdoctor" />}
   138→          {screen === 'weather' && <WeatherScreen key="weather" />}
   139→          {screen === 'insights' && <InsightsScreen key="insights" />}
   140→          {screen === 'alerts' && <AlertsScreen key="alerts" />}
   141→          {screen === 'settings' && <SettingsScreen key="settings" />}
   142→        </AnimatePresence>
   143→      </div>
   144→    </AppContext.Provider>
   145→  );
   146→}
   147→
   148→// ============================================================
   149→// Shared Animation Variants
   150→// ============================================================
   151→const pageVariants = {
   152→  initial: { opacity: 0, y: 20 },
   153→  animate: { opacity: 1, y: 0, transition: { duration: 0.3, ease: 'easeOut' } },
   154→  exit: { opacity: 0, y: -20, transition: { duration: 0.2, ease: 'easeIn' } },
   155→};
   156→
   157→function PageWrapper({ children, className = '' }: { children: React.ReactNode; className?: string }) {
   158→  return (
   159→    <motion.div
   160→      variants={pageVariants}
   161→      initial="initial"
   162→      animate="animate"
   163→      exit="exit"
   164→      className={`min-h-screen ${className}`}
   165→    >
   166→      {children}
   167→    </motion.div>
   168→  );
   169→}

   253→// PROFILE SETUP SCREEN
   254→// ============================================================
   255→function ProfileScreen() {
   256→  const { t, navigate, setProfile } = useApp();
   257→  const [name, setName] = useState('');
   258→  const [village, setVillage] = useState('');
   259→  const [farmSize, setFarmSize] = useState('');
   260→  const [mainCrop, setMainCrop] = useState('');
   261→  const [errors, setErrors] = useState<{ name?: string; village?: string }>({});
   262→
   263→  const farmSizes = [
   264→    { value: '<1', label: t.lessThanOneAcre },
   265→    { value: '1-5', label: t.oneToFiveAcres },
   266→    { value: '5-10', label: t.fiveToTenAcres },
   267→    { value: '>10', label: t.moreThanTenAcres },
   268→  ];
   269→
   270→  const crops = [t.cropWheat, t.cropRice, t.cropCotton, t.cropSoybean, t.cropSugarcane, t.cropTomato, t.cropOnion, t.cropOther];
   271→
   272→  const handleSave = () => {
   273→    const newErrors: { name?: string; village?: string } = {};
   274→    if (!name.trim()) newErrors.name = t.nameRequired;
   275→    if (!village.trim()) newErrors.village = t.villageRequired;
   276→    if (Object.keys(newErrors).length > 0) {
   277→      setErrors(newErrors);
   278→      return;
   279→    }
   280→    setProfile({
   281→      name: name.trim(),
   282→      village: village.trim(),
   283→      farmSize: farmSize || t.lessThanOneAcre,
   284→      mainCrop: mainCrop || t.cropWheat,
   285→    });
   286→    navigate('dashboard');
   287→  };
   288→
   289→  return (
   290→    <PageWrapper className="flex items-center justify-center px-4 py-8">
   291→      <div className="w-full max-w-md">
   292→        <div className="text-center mb-8">
   293→          <div className="w-14 h-14 rounded-2xl bg-gradient-to-br from-green-600 to-emerald-700 flex items-center justify-center shadow-lg mx-auto mb-3">
   294→            <span className="text-2xl">👨‍🌾</span>
   295→          </div>
   296→          <h1 className="text-2xl font-bold text-gray-800">{t.setupProfile}</h1>
   297→          <p className="text-gray-500 mt-1">{t.setupProfileSubtitle}</p>
   298→        </div>
   299→
   300→        <div className="bg-white rounded-3xl shadow-lg p-6 md:p-8 space-y-5">
   301→          {/* Farmer Name */}
   302→          <div>
   303→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.farmerName}</label>
   304→            <input
   305→              type="text"
   306→              value={name}
   307→              onChange={(e) => { setName(e.target.value); setErrors(p => ({ ...p, name: undefined })); }}
   308→              placeholder={t.farmerNamePlaceholder}
   309→              className={`w-full px-4 py-3 rounded-xl border-2 ${errors.name ? 'border-red-400 bg-red-50' : 'border-gray-200 bg-gray-50'} focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800`}
   310→            />
   311→            {errors.name && <p className="text-red-500 text-sm mt-1">{errors.name}</p>}
   312→          </div>
   313→
   314→          {/* Village Name */}
   315→          <div>
   316→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.villageName}</label>
   317→            <input
   318→              type="text"
   319→              value={village}
   320→              onChange={(e) => { setVillage(e.target.value); setErrors(p => ({ ...p, village: undefined })); }}
   321→              placeholder={t.villageNamePlaceholder}
   322→              className={`w-full px-4 py-3 rounded-xl border-2 ${errors.village ? 'border-red-400 bg-red-50' : 'border-gray-200 bg-gray-50'} focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800`}
   323→            />
   324→            {errors.village && <p className="text-red-500 text-sm mt-1">{errors.village}</p>}
   325→          </div>
   326→
   327→          {/* Farm Size */}
   328→          <div>
   329→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.farmSize}</label>
   330→            <div className="grid grid-cols-2 gap-2">
   331→              {farmSizes.map((fs) => (
   332→                <button
   333→                  key={fs.value}
   334→                  onClick={() => setFarmSize(fs.label)}
   335→                  className={`px-3 py-2.5 rounded-xl text-sm font-medium border-2 transition-all ${
   336→                    farmSize === fs.label
   337→                      ? 'border-green-500 bg-green-50 text-green-700'
   338→                      : 'border-gray-200 bg-gray-50 text-gray-600 hover:border-gray-300'
   339→                  }`}
   340→                >
   341→                  {fs.label}
   342→                </button>
   343→              ))}
   344→            </div>
   345→          </div>
   346→
   347→          {/* Main Crop */}
   348→          <div>
   349→            <label className="block text-sm font-semibold text-gray-700 mb-1.5">{t.mainCrop}</label>
   350→            <select
   351→              value={mainCrop}
   352→              onChange={(e) => setMainCrop(e.target.value)}
   353→              className="w-full px-4 py-3 rounded-xl border-2 border-gray-200 bg-gray-50 focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800"
   354→            >
   355→              <option value="">{t.selectCrop}</option>
   356→              {crops.map((c) => (
   357→                <option key={c} value={c}>{c}</option>
   358→              ))}
   359→            </select>
   360→          </div>
   361→
   362→          {/* Buttons */}
   363→          <div className="flex gap-3 pt-2">
   364→            <button
   365→              onClick={() => navigate('language')}
   366→              className="flex-1 py-3 px-4 rounded-xl border-2 border-gray-200 text-gray-600 font-semibold hover:bg-gray-50 transition-all"
   367→            >
   368→              {t.back}
   369→            </button>
   370→            <button
   371→              onClick={handleSave}
   372→              className="flex-1 py-3 px-4 rounded-xl bg-gradient-to-r from-green-600 to-emerald-600 text-white font-semibold shadow-md hover:shadow-lg hover:from-green-700 hover:to-emerald-700 transition-all"
   373→            >
   374→              {t.saveAndContinue}
   375→            </button>
   376→          </div>
   377→        </div>
   378→      </div>
   379→    </PageWrapper>
   380→  );
   381→}

   518→// ============================================================
   519→interface ChatMessage {
   520→  id: number;
   521→  role: 'user' | 'bot';
   522→  text: string;
   523→}
   524→
   525→function ChatbotScreen() {
   526→  const { t, lang } = useApp();
   527→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   528→  const [input, setInput] = useState('');
   529→  const [isTyping, setIsTyping] = useState(false);
   530→  const chatEndRef = React.useRef<HTMLDivElement>(null);
   531→
   532→  useEffect(() => {
   533→    chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   534→  }, [messages, isTyping]);
   535→
   536→  const generateResponse = (userMsg: string): string => {
   537→    const lower = userMsg.toLowerCase();
   538→    if (lower.includes('water') || lower.includes('irrigat') || lower.includes('watering') || lower.includes('पानी') || lower.includes('सिंचाई')) {
   539→      return t.waterResponse;
   540→    }
   541→    if (lower.includes('yellow') || lower.includes('पील') || lower.includes('पिवळ')) {
   542→      return t.yellowLeavesResponse;
   543→    }
   544→    if (lower.includes('pest') || lower.includes('insect') || lower.includes('कीट') || lower.includes('bug')) {
   545→      return t.pestResponse;
   546→    }
   547→    if (lower.includes('fertilizer') || lower.includes('nutrient') || lower.includes('उर्वरक') || lower.includes('पोषक') || lower.includes('खत')) {
   548→      return t.fertilizerResponse;
   549→    }
   550→    if (lower.includes('weather') || lower.includes('मौसम') || lower.includes('हवामान')) {
   551→      return t.weatherResponse;
   552→    }
   553→    return t.defaultResponse;
   554→  };
   555→
   556→  const sendMessage = (text?: string) => {
   557→    const msgText = text || input.trim();
   558→    if (!msgText) return;
   559→
   560→    const userMsg: ChatMessage = { id: Date.now(), role: 'user', text: msgText };
   561→    setMessages(prev => [...prev, userMsg]);
   562→    setInput('');
   563→    setIsTyping(true);
   564→
   565→    setTimeout(() => {
   566→      const botReply: ChatMessage = { id: Date.now() + 1, role: 'bot', text: generateResponse(msgText) };
   567→      setIsTyping(false);
   568→      setMessages(prev => [...prev, botReply]);
   569→    }, 800 + Math.random() * 700);
   570→  };
   571→
   572→  const handleKeyDown = (e: React.KeyboardEvent) => {
   573→    if (e.key === 'Enter' && !e.shiftKey) {
   574→      e.preventDefault();
   575→      sendMessage();
   576→    }
   577→  };
   578→
   579→  return (
   580→    <PageWrapper className="flex flex-col">
   581→      {/* Header */}
   582→      <header className="sticky top-0 z-50 bg-white/80 backdrop-blur-xl border-b border-green-100">
   583→        <div className="max-w-2xl mx-auto px-4 py-3 flex items-center gap-3">
   584→          <BackButton target="dashboard" label={t.backToDashboard} />
   585→          <div className="flex-1 flex items-center gap-2 justify-center">
   586→            <span className="text-xl">🤖</span>
   587→            <h1 className="text-lg font-bold text-green-800">{t.aiAssistant}</h1>
   588→          </div>
   589→        </div>
   590→      </header>
   591→
   592→      {/* Messages */}
   593→      <div className="flex-1 max-w-2xl mx-auto w-full px-4 py-4 overflow-y-auto">
   594→        {messages.length === 0 && (
   595→          <div className="text-center py-12">
   596→            <div className="text-5xl mb-4">🤖</div>
   597→            <h3 className="text-xl font-bold text-gray-800 mb-2">{t.aiAssistant}</h3>
   598→            <p className="text-gray-500 text-sm mb-8 max-w-sm mx-auto">{t.defaultResponse}</p>
   599→
   600→            {/* Quick Suggestions */}
   601→            <div className="space-y-2">
   602→              <p className="text-sm font-semibold text-gray-600 mb-3">{t.quickSuggestions}</p>
   603→              {[t.suggestion1, t.suggestion2, t.suggestion3, t.suggestion4].map((s, i) => (
   604→                <motion.button
   605→                  key={i}
   606→                  initial={{ opacity: 0, y: 10 }}
   607→                  animate={{ opacity: 1, y: 0 }}
   608→                  transition={{ delay: i * 0.1 }}
   609→                  onClick={() => sendMessage(s)}
   610→                  className="w-full text-left px-4 py-3 bg-white rounded-xl border border-green-200 text-sm text-green-800 hover:bg-green-50 hover:border-green-300 transition-all shadow-sm"
   611→                >
   612→                  {s}
   613→                </motion.button>
   614→              ))}
   615→            </div>
   616→          </div>
   617→        )}
   618→
   619→        {/* Message List */}
   620→        {messages.map((msg) => (
   621→          <motion.div
   622→            key={msg.id}
   623→            initial={{ opacity: 0, y: 10 }}
   624→            animate={{ opacity: 1, y: 0 }}
   625→            className={`flex mb-4 ${msg.role === 'user' ? 'justify-end' : 'justify-start'}`}
   626→          >
   627→            <div className={`max-w-[85%] rounded-2xl px-4 py-3 text-sm leading-relaxed ${
   628→              msg.role === 'user'
   629→                ? 'bg-gradient-to-r from-green-600 to-emerald-600 text-white rounded-br-md'
   630→                : 'bg-white border border-gray-200 text-gray-800 shadow-sm rounded-bl-md'
   631→            }`}>
   632→              <div className="whitespace-pre-wrap break-words">{msg.text}</div>
   633→            </div>
   634→          </motion.div>
   635→        ))}
   636→
   637→        {isTyping && (
   638→          <div className="flex mb-4 justify-start">
   639→            <div className="bg-white border border-gray-200 rounded-2xl rounded-bl-md px-4 py-3 shadow-sm">
   640→              <div className="flex gap-1.5">
   641→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '0ms' }} />
   642→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '150ms' }} />
   643→                <div className="w-2 h-2 bg-green-500 rounded-full animate-bounce" style={{ animationDelay: '300ms' }} />
   644→              </div>
   645→            </div>
   646→          </div>
   647→        )}
   648→        <div ref={chatEndRef} />
   649→      </div>
   650→
   651→      {/* Input */}
   652→      <div className="sticky bottom-0 bg-white/80 backdrop-blur-xl border-t border-green-100">
   653→        <div className="max-w-2xl mx-auto px-4 py-3">
   654→          <div className="flex gap-2">
   655→            <input
   656→              type="text"
   657→              value={input}
   658→              onChange={(e) => setInput(e.target.value)}
   659→              onKeyDown={handleKeyDown}
   660→              placeholder={t.typeMessage}
   661→              className="flex-1 px-4 py-3 rounded-xl border-2 border-gray-200 bg-gray-50 focus:border-green-500 focus:bg-white focus:outline-none transition-all text-gray-800 text-sm"
   662→            />
   663→            <button
   664→              onClick={() => sendMessage()}
   665→              disabled={!input.trim() || isTyping}
   666→              className="px-5 py-3 rounded-xl bg-gradient-to-r from-green-600 to-emerald-600 text-white font-semibold shadow-md hover:shadow-lg hover:from-green-700 hover:to-emerald-700 disabled:opacity-50 disabled:cursor-not-allowed transition-all"
   667→            >
   668→              {t.send}
   669→            </button>
   670→          </div>
   671→        </div>
   672→      </div>
   673→    </PageWrapper>
   674→  );
   675→}

   684→  const [result, setResult] = useState<boolean>(false);
   685→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   686→
   687→  const handleFileSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   688→    const file = e.target.files?.[0];
   689→    if (file) {
   690→      const reader = new FileReader();
   691→      reader.onload = (ev) => {
   692→        setSelectedImage(ev.target?.result as string);
   693→        setResult(false);
   694→      };

   696→    }
   697→  };

   700→    setAnalyzing(true);
   701→    setTimeout(() => {
   702→      setAnalyzing(false);
   703→      setResult(true);

   705→  };

   707→  const handleAnalyzeAnother = () => {
   708→    setSelectedImage(null);
   709→    setResult(false);
   710→    if (fileInputRef.current) fileInputRef.current.value = '';
   711→  };

   713→  return (

   715→      <header className="sticky top-0 z-50 bg-white/80 backdrop-blur-xl border-b border-green-100">
