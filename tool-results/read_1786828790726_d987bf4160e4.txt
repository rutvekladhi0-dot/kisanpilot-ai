    84→    84→  });
    85→    85→  const [screen, setScreen] = useState<Screen>(() => {
    86→    86→    if (typeof window !== 'undefined') {
    87→    87→      const savedLang = localStorage.getItem('kp_lang');

    89→    89→      if (savedLang && savedProfile) return 'dashboard';
    90→    90→      if (savedLang) return 'profile';
    91→    91→    }
    92→    92→    return 'language';
    93→    93→  });
    94→    94→  const [alerts, setAlertsState] = useState<AlertItem[]>(() => {
    95→    95→    if (typeof window !== 'undefined') {
    96→    96→      try {
    97→    97→        const saved = localStorage.getItem('kp_alerts');
    98→    98→        return saved ? JSON.parse(saved) : defaultAlerts;
    99→    99→      } catch { return defaultAlerts; }
   100→   100→    }
   101→   101→    return defaultAlerts;
   102→   102→  });
   103→   103→
   104→   104→  const [mounted, setMounted] = useState(false);
   105→   105→
   106→   106→  useEffect(() => {
   107→   107→    if (!initialized.current) {
   108→   108→      initialized.current = true;
   109→   109→      requestAnimationFrame(() => setMounted(true));
   110→   110→    }
   111→   111→  }, []);
   112→   112→
   113→   113→  const t = getTranslations(lang);
   114→   114→
   115→   115→  const setLang = useCallback((l: Language) => {
   116→   116→    setLangState(l);
   117→   117→    localStorage.setItem('kp_lang', l);
   118→   118→  }, []);
   119→   119→
   120→   120→  const setProfile = useCallback((p: FarmerProfile) => {
   121→   121→    setProfileState(p);
   122→   122→    localStorage.setItem('kp_profile', JSON.stringify(p));
   123→   123→  }, []);
   124→   124→
   125→   125→  const navigate = useCallback((s: Screen) => {
   126→   126→    setScreen(s);
   127→   127→    window.scrollTo(0, 0);
   128→   128→  }, []);
   129→   129→
   130→   130→  const setAlerts = useCallback((a: AlertItem[]) => {
   131→   131→    setAlertsState(a);
   132→   132→    localStorage.setItem('kp_alerts', JSON.stringify(a));
   133→   133→  }, []);
   134→   134→
   135→   135→  if (!mounted) {
   136→   136→    return null;
   137→   137→  }
   138→   138→
   139→   139→  return (
   140→   140→    <AppContext.Provider value={{ lang, t, setLang, profile, setProfile, screen, navigate, alerts, setAlerts }}>
   141→   141→      <div className="min-h-screen bg-gradient-to-br from-green-50 via-white to-emerald-50">
   142→   142→        <AnimatePresence mode="wait">
   143→   143→          {screen === 'language' && <LanguageScreen key="language" />}
   144→   144→          {screen === 'profile' && <ProfileScreen key="profile" />}
   145→   145→          {screen === 'dashboard' && <DashboardScreen key="dashboard" />}
   146→   146→          {screen === 'chatbot' && <ChatbotScreen key="chatbot" />}
   147→   147→          {screen === 'myfarm' && <MyFarmScreen key="myfarm" />}
   148→   148→          {screen === 'cropdoctor' && <CropDoctorScreen key="cropdoctor" />}
   149→   149→          {screen === 'weather' && <WeatherScreen key="weather" />}
   150→   150→          {screen === 'insights' && <InsightsScreen key="insights" />}
   151→   151→          {screen === 'alerts' && <AlertsScreen key="alerts" />}
   152→   152→          {screen === 'settings' && <SettingsScreen key="settings" />}
   153→   153→          {screen === 'voicekhata' && <VoiceKhataScreen key="voicekhata" />}
   154→   154→          {screen === 'farmeconomics' && <FarmEconomicsScreen key="farmeconomics" />}
   155→   155→          {screen === 'bajarbhav' && <BajarBhavScreen key="bajarbhav" />}
   156→   156→          {screen === 'riskradar' && <RiskRadarScreen key="riskradar" />}
   157→   157→          {screen === 'simulator' && <SimulatorScreen key="simulator" />}
   158→   158→          {screen === 'pmfby' && <PMFBYScreen key="pmfby" />}
   159→   159→          {screen === 'farmgoals' && <FarmGoalsScreen key="farmgoals" />}
   160→   160→          {screen === 'nextbestaction' && <NextBestActionScreen key="nextbestaction" />}
   161→   161→          {screen === 'phototracker' && <MonthlyPhotoTrackerScreen key="phototracker" />}
   162→   162→          {screen === 'farmmemory' && <FarmMemoryScreen key="farmmemory" />}
   163→   163→          {screen === 'seasonscorecard' && <SeasonScoreCardScreen key="seasonscorecard" />}

   277→   277→function ProfileScreen() {
   278→   278→  const { t, navigate, setProfile } = useApp();
   279→   279→  const [name, setName] = useState('');
   280→   280→  const [village, setVillage] = useState('');
   281→   281→  const [farmSize, setFarmSize] = useState('');
   282→   282→  const [mainCrop, setMainCrop] = useState('');
   283→   283→  const [photo, setPhoto] = useState<string | null>(null);
   284→   284→  const [errors, setErrors] = useState<{ name?: string; village?: string }>({});
   285→   285→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   286→   286→
   287→   287→  const farmSizes = [
   288→   288→    { value: '<1', label: t.lessThanOneAcre },
   289→   289→    { value: '1-5', label: t.oneToFiveAcres },
   290→   290→    { value: '5-10', label: t.fiveToTenAcres },

   292→   292→  ];
   293→   293→
   294→   294→  const crops = [t.cropWheat, t.cropRice, t.cropCotton, t.cropSoybean, t.cropSugarcane, t.cropTomato, t.cropOnion, t.cropOther];
   295→   295→
   296→   296→  const handleSave = () => {
   297→   297→    const newErrors: { name?: string; village?: string } = {};
   298→   298→    if (!name.trim()) newErrors.name = t.nameRequired;
   299→   299→    if (!village.trim()) newErrors.village = t.villageRequired;
   300→   300→    if (Object.keys(newErrors).length > 0) {
   301→   301→      setErrors(newErrors);

   303→   303→    }

   305→   305→      name: name.trim(),
   306→   306→      village: village.trim(),
   307→   307→      farmSize: farmSize || t.lessThanOneAcre,
   308→   308→      mainCrop: mainCrop || t.cropWheat,

   310→   310→    if (photo) {
   311→   311→      localStorage.setItem('kp_farmer_photo', photo);
   312→   312→    }
   313→   313→    navigate('dashboard');

   353→   353→              {t.changePhoto}

   664→   664→interface ChatMessage {
   665→   665→  id: number;
   666→   666→  role: 'user' | 'bot';
   667→   667→  text: string;
   668→   668→}
   669→   669→
   670→   670→function ChatbotScreen() {
   671→   671→  const { t, lang } = useApp();
   672→   672→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   673→   673→  const [input, setInput] = useState('');
   674→   674→  const [isTyping, setIsTyping] = useState(false);
   675→   675→  const [voiceState, setVoiceState] = useState<'idle' | 'listening'>('idle');
   676→   676→  const [liveTranscript, setLiveTranscript] = useState('');
   677→   677→  const chatEndRef = React.useRef<HTMLDivElement>(null);
   678→   678→  const recognitionRef = React.useRef<any>(null);
   679→   679→
   680→   680→  useEffect(() => {
   681→   681→    chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   682→   682→  }, [messages, isTyping]);
   683→   683→
   684→   684→  const generateResponse = (userMsg: string): string => {
   685→   685→    const lower = userMsg.toLowerCase();
   686→   686→    if (lower.includes('water') || lower.includes('irrigat') || lower.includes('watering') || lower.includes('पानी') || lower.includes('सिंचाई')) {
   687→   687→      return t.waterResponse;
   688→   688→    }
   689→   689→    if (lower.includes('yellow') || lower.includes('पील') || lower.includes('पिवळ')) {
   690→   690→      return t.yellowLeavesResponse;
   691→   691→    }
   692→   692→    if (lower.includes('pest') || lower.includes('insect') || lower.includes('कीट') || lower.includes('bug')) {
   693→   693→      return t.pestResponse;
   694→   694→    }
   695→   695→    if (lower.includes('fertilizer') || lower.includes('nutrient') || lower.includes('उर्वरक') || lower.includes('पोषक') || lower.includes('खत')) {
   696→   696→      return t.fertilizerResponse;
   697→   697→    }
   698→   698→    if (lower.includes('weather') || lower.includes('मौसम') || lower.includes('हवामान')) {
   699→   699→      return t.weatherResponse;
   700→   700→    }
   701→   701→    return t.defaultResponse;
   702→   702→  };
   703→   703→
   704→   704→  const startVoice = () => {
   705→   705→    const SpeechRecognition = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
   706→   706→    if (!SpeechRecognition) return;
   707→   707→    const recognition = new SpeechRecognition();
   708→   708→    recognition.lang = lang === 'hi' ? 'hi-IN' : lang === 'mr' ? 'mr-IN' : 'en-IN';
   709→   709→    recognition.continuous = true;
   710→   710→    recognition.interimResults = true;
   711→   711→    recognition.onresult = (event: any) => {
   712→   712→      let final = '';
   713→   713→      let interim = '';
   714→   714→      for (let i = event.resultIndex; i < event.results.length; i++) {
   715→   715→        if (event.results[i].isFinal) {
   716→   716→          final += event.results[i][0].transcript;
   717→   717→        } else {
   718→   718→          interim += event.results[i][0].transcript;
   719→   719→        }
   720→   720→      }
   721→   721→      if (final) {
   722→   722→        setInput(final);
   723→   723→        setLiveTranscript('');
   724→   724→      } else if (interim) {
   725→   725→        setInput(interim);
   726→   726→        setLiveTranscript(interim);
   727→   727→      }
   728→   728→    };
   729→   729→    recognition.onerror = () => {
   730→   730→      setVoiceState('idle');
   731→   731→    };
   732→   732→    recognition.onend = () => {
   733→   733→      if (voiceState === 'listening') {
   734→   734→        recognition.start();
   735→   735→      }
   736→   736→    };
   737→   737→    recognitionRef.current = recognition;
   738→   738→    recognition.start();
   739→   739→    setVoiceState('listening');
   740→   740→  };
   741→   741→
   742→   742→  const stopVoice = () => {
   743→   743→    if (recognitionRef.current) {
   744→   744→      recognitionRef.current.stop();
   745→   745→      recognitionRef.current = null;
   746→   746→    }
   747→   747→    setVoiceState('idle');
   748→   748→    setLiveTranscript('');
   749→   749→    if (input.trim()) {
   750→   750→      sendMessage();
   751→   751→    }
   752→   752→  };
   753→   753→
   754→   754→  const toggleVoice = () => {
   755→   755→    if (voiceState === 'listening') {
   756→   756→      stopVoice();
   757→   757→    } else {
   758→   758→      startVoice();
   759→   759→    }
   760→   760→  };
   761→   761→
